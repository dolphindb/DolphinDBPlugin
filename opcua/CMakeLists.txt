cmake_minimum_required(VERSION 3.22)

set(PluginName PluginOPCUA)
project(${PluginName})

include(../template.cmake)
CreatePlugin(${PluginName})

if (WIN32 OR MINGW)
    target_compile_definitions(${PluginName} PRIVATE OPCUA)
    target_link_directories(${PluginName} PRIVATE ${CMAKE_CURRENT_LIST_DIR}/lib/)
    target_link_libraries(PluginOPCUA PRIVATE libmbedcrypto.a libmbedx509.a libmbedtls.a open62541)
    cmake_path(GET CMAKE_CURRENT_SOURCE_DIR FILENAME INSTALL_DIR)
    install(FILES ${CMAKE_CURRENT_LIST_DIR}/lib/libopen62541.dll DESTINATION ${INSTALL_DIR})
elseif (UNIX)
    target_link_options(${PluginName} PRIVATE "-flto")
    AddDependency(open62541 open62541)
    AddDependency(mbedtls mbedcrypto)
    AddDependency(mbedtls mbedx509)
endif ()
